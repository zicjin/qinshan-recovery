CREATE OR REPLACE PROCEDURE INSERTI18N(CODEPARAM      VARCHAR,
                                       DESCRIPTION    VARCHAR,
                                       DEFAULTMESSAGE VARCHAR,
                                       LANG           VARCHAR,
                                       PMESSAGE        VARCHAR) AS
  V_ID     NUMBER;
  V_COUNT  NUMBER;
  V_MCOUNT NUMBER;
BEGIN

  V_COUNT  := 0;
  V_MCOUNT := 0;

  SELECT COUNT(*) INTO V_COUNT FROM T_CORE_I18N WHERE CODE = CODEPARAM;

  IF V_COUNT > 0 THEN
    SELECT ID
      INTO V_ID
      FROM T_CORE_I18N
     WHERE CODE = CODEPARAM
       AND ROWNUM = 1;

    SELECT COUNT(*)
      INTO V_MCOUNT
      FROM T_CORE_I18N_MESSAGE
     WHERE I18NID = V_ID
       AND I18NCONFIGID = LANG;

    IF V_MCOUNT > 0 THEN
      UPDATE T_CORE_I18N_MESSAGE
         SET MESSAGE = PMESSAGE
       WHERE I18NID = V_ID
         AND I18NCONFIGID = LANG;
    END IF;

  ELSE
    SELECT SEQ_CORE.NEXTVAL INTO V_ID FROM DUAL;
    INSERT INTO T_CORE_I18N
      (ID, CODE, DESCRIPTION, DEFAULTMESSAGE)
    VALUES
      (V_ID, CODEPARAM, DESCRIPTION, DEFAULTMESSAGE);
  END IF;

  IF V_MCOUNT = 0 THEN
    INSERT INTO T_CORE_I18N_MESSAGE
      (ID, I18NID, I18NCONFIGID, MESSAGE)
    VALUES
      (SEQ_CORE.NEXTVAL, V_ID, LANG, PMESSAGE);
  END IF;
  COMMIT;
END;
