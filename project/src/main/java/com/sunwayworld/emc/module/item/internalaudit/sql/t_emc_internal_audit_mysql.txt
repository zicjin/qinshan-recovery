







DROP TABLE IF EXISTS `t_emc_internal_audit`;
CREATE TABLE `t_emc_internal_audit`  (
  `id` bigint(32) NOT NULL COMMENT '主键',
	`qualityControlNo` varchar(18) CHARACTER SET utf8 COLLATE utf8_general_ci NULL DEFAULT NULL COMMENT '质控编号',
  `auditPurpose` varchar(540) CHARACTER SET utf8 COLLATE utf8_general_ci NULL DEFAULT NULL COMMENT '审核目的',
  `auditBasis` varchar(540) CHARACTER SET utf8 COLLATE utf8_general_ci NULL DEFAULT NULL COMMENT '审核依据',
  `auditOrg` varchar(270) CHARACTER SET utf8 COLLATE utf8_general_ci NULL DEFAULT NULL COMMENT '被审核部门',
  `auditGroupUser` varchar(360) CHARACTER SET utf8 COLLATE utf8_general_ci NULL DEFAULT NULL COMMENT '内审组成员',
  `auditTime` datetime NULL DEFAULT NULL COMMENT '拟计划审核时间',
  `auditContent` varchar(540) CHARACTER SET utf8 COLLATE utf8_general_ci NULL DEFAULT NULL COMMENT '拟计划审核内容',
  `userTrain` varchar(540) CHARACTER SET utf8 COLLATE utf8_general_ci NULL DEFAULT NULL COMMENT '内审人员培训（必要时）',
  `recordTable` varchar(540) CHARACTER SET utf8 COLLATE utf8_general_ci NULL DEFAULT NULL COMMENT '记录表格',
  `startTime` datetime NULL DEFAULT NULL COMMENT '内审开始时间',
  `endTime` datetime NULL DEFAULT NULL COMMENT '内审结束时间',
  `auditDate` datetime NULL DEFAULT NULL COMMENT '审核日期',
  `lastAuditDate` datetime NULL DEFAULT NULL COMMENT '上次审核日期',
  `division` varchar(540) CHARACTER SET utf8 COLLATE utf8_general_ci NULL DEFAULT NULL COMMENT '审核成员分工及审核部门',
  `auditSituation` varchar(540) CHARACTER SET utf8 COLLATE utf8_general_ci NULL DEFAULT NULL COMMENT '审核情况综述',
  `inconformity` varchar(540) CHARACTER SET utf8 COLLATE utf8_general_ci NULL DEFAULT NULL COMMENT '不符合项目归纳分类及分布项',
  `manageSystem` varchar(540) CHARACTER SET utf8 COLLATE utf8_general_ci NULL DEFAULT NULL COMMENT '受审方的管理体系执行情况的整体综合评价',
  `correct` varchar(540) CHARACTER SET utf8 COLLATE utf8_general_ci NULL DEFAULT NULL COMMENT '纠正措施或预防措施执行情况',
  `auditRequest` varchar(540) CHARACTER SET utf8 COLLATE utf8_general_ci NULL DEFAULT NULL COMMENT '审核后续活动要求',
  `auditReview` varchar(540) CHARACTER SET utf8 COLLATE utf8_general_ci NULL DEFAULT NULL COMMENT '审核综述',
  `sendScope` varchar(540) CHARACTER SET utf8 COLLATE utf8_general_ci NULL DEFAULT NULL COMMENT '发送范围',
   `CREATEDBYID` varchar(32) DEFAULT NULL COMMENT '制单人ID',
  `CREATEDBYNAME` varchar(64) DEFAULT NULL COMMENT '制单人名称',
  `CREATEDTIME` datetime  DEFAULT NULL COMMENT '制单日期',
  `CREATEDBYORGID` varchar(32) DEFAULT NULL COMMENT '制单人单位编码',
  `CREATEDBYORGNAME` varchar(256) DEFAULT NULL COMMENT '制单人单位名称',
	`PROCESSSTATUS` VARCHAR(36)  DEFAULT NULL COMMENT '流程状态',
  PRIMARY KEY (`id`) USING BTREE
) ENGINE = InnoDB CHARACTER SET = utf8 COLLATE = utf8_general_ci ROW_FORMAT = Dynamic;
