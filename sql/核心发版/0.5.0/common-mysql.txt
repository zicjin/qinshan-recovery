UPDATE T_CORE_REPORT_CONFIG A,
( SELECT ( @RN := @RN + 1 ) RN, ID FROM T_CORE_REPORT_CONFIG WHERE ORDERNO IS NULL ORDER BY ID ) B,
( SELECT @RN := 0 ) RN 
SET A.ORDERNO = B.RN 
WHERE A.ID = B.ID;

UPDATE T_CORE_REPORT_CONFIG_LINE A,
( SELECT ( @RN := @RN + 1 ) RN, ID FROM T_CORE_REPORT_CONFIG_LINE WHERE ORDERNO IS NULL ORDER BY REPORTCONFIGID,ID ) B,
( SELECT @RN := 0 ) RN 
SET A.ORDERNO = B.RN 
WHERE A.ID = B.ID;
