-- INSERTI18N
CREATE OR REPLACE PROCEDURE "public"."inserti18n"("codeparam" varchar, "description" varchar, "defaultmessage" varchar, "lang" varchar, "pmessage" varchar)
 AS $BODY$
  Declare V_ID     int8;
  Declare V_COUNT  int8;
  Declare V_MCOUNT int8;
BEGIN
    V_ID := NULL;
    SELECT
        ID INTO V_ID 
    FROM
        T_CORE_I18N 
    WHERE
        CODE = CODEPARAM;
    IF
        ( V_ID IS NULL ) THEN
        SELECT
            NEXTVAL('SEQ_CORE') INTO V_ID;
        INSERT INTO T_CORE_I18N ( ID, CODE, DESCRIPTION, DEFAULTMESSAGE )
        VALUES
            ( V_ID, CODEPARAM, DESCRIPTION, DEFAULTMESSAGE );
        
    END IF;
    INSERT INTO T_CORE_I18N_MESSAGE ( ID, I18NID, I18NCONFIGID, MESSAGE )
    VALUES
        ( NEXTVAL('SEQ_CORE'), V_ID, LANG, PMESSAGE );
    COMMIT;
END;
$BODY$
  LANGUAGE plpgsql;

-- UPDATEI18N
CREATE OR REPLACE PROCEDURE "public"."updatei18n"("codeparam" varchar, "lang" varchar, "messageinfo" varchar)
 AS $BODY$
BEGIN
  UPDATE T_CORE_I18N_MESSAGE
     SET message = MESSAGEINFO
   WHERE EXISTS (SELECT 1
            FROM T_CORE_I18N I
           WHERE I.ID = I18NID
             AND I.CODE = CODEPARAM
             AND I18NCONFIGID = LANG);

  COMMIT;
END;
$BODY$
  LANGUAGE plpgsql;

-- DELETEI18N
CREATE OR REPLACE PROCEDURE "public"."deletei18n"("codeparam" varchar)
 AS $BODY$
  Declare V_ID    int8;
  Declare V_COUNT int8;
BEGIN

    V_ID := NULL;
    SELECT
        ID INTO V_ID 
    FROM
        T_CORE_I18N 
    WHERE
        CODE = CODEPARAM;
    IF
        ( V_ID IS NOT NULL ) THEN
        DELETE FROM T_CORE_I18N_MESSAGE WHERE I18NID = V_ID;
        DELETE FROM T_CORE_I18N WHERE CODE = CODEPARAM;
        COMMIT;
        
    END IF;
END;
$BODY$
  LANGUAGE plpgsql;
  
-- sequence SEQ_CORE
create sequence SEQ_CORE
minvalue 1
maxvalue 999999999999999
start with 20000
increment by 1
cache 1000
cycle;
