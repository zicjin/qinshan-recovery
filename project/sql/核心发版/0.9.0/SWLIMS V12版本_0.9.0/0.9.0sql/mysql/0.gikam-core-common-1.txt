DROP PROCEDURE IF EXISTS `DELETEI18N`;
DELIMITER $$
-- Automation Deployment Single SQL ---begin
CREATE PROCEDURE `DELETEI18N` (
    CODEPARAM VARCHAR ( 128 )
    ) 
BEGIN
    
    SET @V_ID = NULL;
    SELECT
        ID INTO @V_ID 
    FROM
        T_CORE_I18N 
    WHERE
        CODE = CODEPARAM;
    IF
        ( @V_ID IS NOT NULL ) THEN
        DELETE FROM T_CORE_I18N_MESSAGE WHERE I18NID = @V_ID;
        DELETE FROM T_CORE_I18N WHERE CODE = CODEPARAM;
        COMMIT;
        
    END IF;
END$$
-- Automation Deployment Single SQL ---end
DELIMITER ;
