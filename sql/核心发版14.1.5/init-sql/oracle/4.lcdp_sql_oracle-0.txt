-- 根据视图名查询视图创建SELECT语句
CREATE OR REPLACE FUNCTION VIEW_SELECT(uVIEW IN VARCHAR2)
RETURN VARCHAR2
AS
  uVal VARCHAR2(4000);
BEGIN
  SELECT TEXT INTO uVal FROM USER_VIEWS WHERE VIEW_NAME = UPPER(uVIEW);
  RETURN uVal;
END VIEW_SELECT;

-- 表字段默认值字段为LONG类型 转为 VARCHAR2
CREATE OR REPLACE FUNCTION LONG_TO_CHAR(uTABLE IN VARCHAR2,uID IN NUMBER)
RETURN VARCHAR2
AS
  uVal VARCHAR2(4000);
BEGIN
  SELECT DATA_DEFAULT INTO uVal FROM USER_TAB_COLS WHERE TABLE_NAME = UPPER(uTABLE) AND COLUMN_ID = uID;
  RETURN uVal;
END LONG_TO_CHAR;
