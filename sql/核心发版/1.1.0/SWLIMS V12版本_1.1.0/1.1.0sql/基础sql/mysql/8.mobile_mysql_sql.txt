DROP TABLE IF EXISTS `T_CORE_MOBILE_MENU`;
CREATE TABLE `T_CORE_MOBILE_MENU` (
  `ID` BIGINT NOT NULL COMMENT '主键',
  `MENUNAME` VARCHAR(64)  DEFAULT NULL COMMENT '菜单名称' ,
  `CREATEDBYID` VARCHAR(32)  DEFAULT NULL COMMENT '制单人编码',
  `CREATEDBYNAME` VARCHAR(64)  DEFAULT NULL COMMENT '制单人名称',
  `CREATEDTIME` DATETIME(0)  DEFAULT NULL COMMENT '制单时间',
  `CREATEDBYORGID` VARCHAR(32)  DEFAULT NULL COMMENT '制单人单位编码',
  `CREATEDBYORGNAME` VARCHAR(256)  DEFAULT NULL COMMENT '制单人单位名称',
  `ACTIVATEDFLAG` VARCHAR(1)  DEFAULT '0' COMMENT '状态:是否启用',
  `REMARK` VARCHAR(256) DEFAULT NULL COMMENT '备注',
  `ACTIVATEDBYID` VARCHAR(32) DEFAULT NULL COMMENT '状态修改人ID',
  `ACTIVATEDBYNAME` VARCHAR(256) DEFAULT NULL COMMENT '状态修改人名称',
  `ICON` VARCHAR(256) DEFAULT NULL COMMENT '图标',
  `BADGECONFIG` TEXT DEFAULT NULL COMMENT '角标配置',
  `ORDERNO` BIGINT(20)  DEFAULT NULL COMMENT '排序号',
  `ACTIVATEDTIME` DATETIME(0)  DEFAULT NULL COMMENT '启用时间',
  `BADGEFLAG` VARCHAR(1)  DEFAULT '0' COMMENT '是否启用角标',
  `CLICKCONFIG` TEXT DEFAULT NULL COMMENT '菜单按钮点击配置',
   PRIMARY KEY (`ID`)
)COMMENT='移动菜单表' ENGINE = InnoDB COLLATE = utf8_general_ci;

DROP TABLE IF EXISTS `T_CORE_MOBILE_PAGE`;
CREATE TABLE `T_CORE_MOBILE_PAGE`  (
  `ID` BIGINT(20) NOT NULL COMMENT '主键',
  `PARENTID` BIGINT(20)  DEFAULT NULL COMMENT '父级ID',
  `PAGENO` VARCHAR(12)   DEFAULT NULL COMMENT '页面编号',
  `PAGENAME` VARCHAR(64)   DEFAULT NULL COMMENT '页面名称',
  `REMARK` TEXT DEFAULT NULL COMMENT '备注',
  `CREATEDBYID` VARCHAR(32)   DEFAULT NULL COMMENT '制单人编码',
  `CREATEDBYNAME` VARCHAR(64)   DEFAULT NULL COMMENT '制单人名称',
  `CREATEDTIME` DATETIME(0)  DEFAULT NULL COMMENT '制单时间',
  `CREATEDBYORGID` VARCHAR(32)   DEFAULT NULL COMMENT '制单人单位编码',
  `CREATEDBYORGNAME` VARCHAR(256)   DEFAULT NULL COMMENT '制单人单位名称',
  `LISTSHOWCONFIG` TEXT DEFAULT NULL COMMENT '列表显示配置',
  `LISTCLICKCONFIG` TEXT DEFAULT NULL COMMENT '列表点击配置',
  `LISTSEARCHCONFIG` TEXT DEFAULT NULL COMMENT '列表搜索配置',
  `CHOOSEBTNCONFIG` TEXT DEFAULT NULL COMMENT '选择按钮配置',
  `MENUBTNCONFIG` TEXT DEFAULT NULL COMMENT '菜单按钮配置',
  `ORDERNO` BIGINT(20)  DEFAULT NULL COMMENT '排序号',
  `LVL` INT(11)  DEFAULT NULL COMMENT '页面等级',
  `MENUID` BIGINT(20)  DEFAULT NULL COMMENT '菜单ID',
   PRIMARY KEY (`ID`)
)COMMENT='移动菜单页面表' ENGINE = InnoDB COLLATE = utf8_general_ci;
